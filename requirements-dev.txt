# Contributor tooling only. See pytest.ini - no plugin depends on this, and
# nothing here reaches a user who installs a plugin.
pytest>=7.0

# Property-based tests, pinned below 7 on purpose. Hypothesis dropped Python
# 3.9 in 6.142.0 (2025-10-16), so pip resolves 6.141.1 on the 3.9 leg and the
# current release on 3.13. Both run the same properties, which means the tests
# must stay on strategies that existed in 6.141.1 - the 3.9 leg fails loudly if
# one does not. Do not "upgrade" this to >=6.142 without also dropping 3.9.
hypothesis>=6.100,<7

# The formatter, pinned exactly. Ruff's formatting output can change between
# releases, and a contributor on a different version would reformat lines nobody
# touched, or fail CI's check on code that is correct for their version. Bump it
# on its own, in a commit that also carries whatever reformatting the new
# version asks for. Configured in ruff.toml.
ruff==0.16.7

# The shell linter, pinned exactly for the same reason: a new release adds
# checks, and a contributor on a different version would see warnings CI does
# not, or miss ones it does. This package is the shellcheck binary itself, so
# nothing else needs installing. Bump it on its own, with any fixes it asks for.
shellcheck-py==0.11.0.1

# Coverage, pinned exactly: the floors in .github/coverage-floors.json are one
# coverage.py version's figures, and another version can count branches
# differently. coverage.py dropped Python 3.9 in 7.11, so the 3.9 leg gets the
# last release that supports it. CI checks the floors on the 3.13 leg only, so
# that pin only has to install and run. Bump the 3.10+ pin on its own, in a
# commit that also resets any floor the new version moves. pytest-cov is what
# runs coverage under `pytest --cov`; .coveragerc configures both.
coverage==7.10.7; python_version < "3.10"
coverage==7.16.1; python_version >= "3.10"
pytest-cov==7.1.0
